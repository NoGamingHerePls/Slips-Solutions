#include<stdlib.h>
#include<string.h>
#include<unistd.h>
#include<sys/wait.h>

int i;
while(1){
printf("$ ");
fgets(command,100,stdin);
command[strlen(command)-1]='\0';
i=0;
args[i]=strtok(command," ");
while(args[i]!=NULL){
i++;
args[i]=strtok(NULL," ");
}
if(strcmp(args[0],"exit")==0) break;
if(strcmp(args[0],"count")==0){
FILE *fp;
char ch;
int count=0;

if(fp==NULL){
printf("File not found\n");
continue;
}
if(args[1][0]=='c'){
while((ch=fgetc(fp))!=EOF) count++;
printf("Characters=%d\n",count);
}
else if(args[1][0]=='w'){
int inword=0;
while((ch=fgetc(fp))!=EOF){
if(ch==' '||ch=='\n'||ch=='\t') inword=0;
else if(inword==0){
count++;
inword=1;
}
}
printf("Words=%d\n",count);
}
else if(args[1][0]=='l'){
while((ch=fgetc(fp))!=EOF)
if(ch=='\n') count++;
printf("Lines=%d\n",count);
}
fclose(fp);
}
else{
if(fork()==0){
execvp(args[0],args);
printf("Command not found\n");
exit(0);
}
else wait(NULL);
}
}
return 0;
}

Q.1) Write a C program to simulate Banker’s algorithm for the purpose of deadlock avoidance. Consider the following snapshot of system, A, B, C and D is the resource type. Process Allocation Max Available  A B C D A B C D A B C D P0 0 0 1 2 0 0 1 2 1 5 2 0 P1 1 0 0 0 1 7 5 0     P2 1 3 5 4 2 3 5 6     P3 0 6 3 2 0 6 5 2     P4 0 0 1 4 0 6 5 6     a)  Calculate and display the content of need matrix? b)  Is the system in safe state? If display the safe sequence.

#include <stdio.h>
int main()
{
    int alloc[5][4] = {
        {0,0,1,2},
        {1,0,0,0},
        {1,3,5,4},
        {0,6,3,2},
        {0,0,1,4}
    };

    int max[5][4] = {
        {0,0,1,2},
        {1,7,5,0},
        {2,3,5,6},
        {0,6,5,2},
        {0,6,5,6}
    };

    int avail[4] = {1,5,2,0};
    int need[5][4], finish[5] = {0};
    int safe[5], count = 0, i, j, k, found;

    /* Calculate Need */
    for(i = 0; i < 5; i++)
        for(j = 0; j < 4; j++)
            need[i][j] = max[i][j] - alloc[i][j];

    printf("Need Matrix:\n");
    printf("     A B C D\n");

    for(i = 0; i < 5; i++)
        printf("P%d : %d %d %d %d\n", i,
               need[i][0], need[i][1],
               need[i][2], need[i][3]);

    /* Banker's Algorithm */
    while(count < 5)
    {
        found = 0;

        for(i = 0; i < 5; i++)
        {
            if(finish[i] == 0)
            {
                for(j = 0; j < 4; j++)
                    if(need[i][j] > avail[j])
                        break;

                if(j == 4)
                {
                    for(k = 0; k < 4; k++)
                        avail[k] += alloc[i][k];

                    safe[count++] = i;
                    finish[i] = 1;
                    found = 1;
                }
            }
        }

        if(found == 0)
            break;
    }

    if(count == 5)
    {
        printf("\nSystem is in SAFE state.\n");
        printf("Safe Sequence: ");

        for(i = 0; i < 5; i++)
        {
            printf("P%d", safe[i]);
            if(i < 4)
                printf(" -> ");
        }
        printf("\n");
    }
    else
        printf("\nSystem is NOT in safe state.\n");

    return 0;
}

Q.2) Write a simulation program for disk scheduling using SCAN algorithm.  Accept total Number of disk blocks, disk request string, and current  head position from the user. Display the list of requests in the  order in which it is served. Also display the total number of head moments.      82, 170, 43, 140, 24, 16, 190, 65      Starting Head position= 50      Direction: Left


#include <stdio.h>
#include <stdlib.h>

int main()
{
    int blocks, n, head, i, j, temp, total = 0;
    int req[20];

    printf("Enter total number of disk blocks: ");
    scanf("%d", &blocks);

    printf("Enter number of requests: ");
    scanf("%d", &n);

    printf("Enter request string:\n");
    for(i = 0; i < n; i++)
        scanf("%d", &req[i]);

    printf("Enter current head position: ");
    scanf("%d", &head);

    /* Sort requests */
    for(i = 0; i < n - 1; i++)
    {
        for(j = i + 1; j < n; j++)
        {
            if(req[i] > req[j])
            {
                temp = req[i];
                req[i] = req[j];
                req[j] = temp;
            }
        }
    }

    printf("\nOrder of service: %d", head);

    /* Move Left */
    for(i = n - 1; i >= 0; i--)
    {
        if(req[i] < head)
        {
            total += abs(req[i] - head);
            head = req[i];
            printf(" -> %d", head);
        }
    }

    /* Move to 0 */
    total += head;
    head = 0;
    printf(" -> %d", head);

    /* Move Right */
    for(i = 0; i < n; i++)
    {
        if(req[i] > head)
        {
            total += abs(req[i] - head);
            head = req[i];
            printf(" -> %d", head);
        }
    }

    printf("\nTotal head movement = %d\n", total);

    return 0;
}

Q.1) Write the simulation program for demand paging and show the page scheduling  total and number of page faults according the FIFO page replacement  algorithm. Assume    the memory of n frames. Reference String: 3,4,5,6,3,4,7,3,4,5,6,7,2,4,6

#include <stdio.h>

int main()
{
    int n, i, j, page, pos = 0;
    int pages[20] = {3,4,5,6,3,4,7,3,4,5,6,7,2,4,6};
    int frame[10];
    int faults = 0, hit;

    printf("Enter number of frames: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++)
        frame[i] = -1;

    printf("\nPage\tFrames\t\tStatus\n");

    for(i = 0; i < 15; i++)
    {
        page = pages[i];
        hit = 0;

        for(j = 0; j < n; j++)
        {
            if(frame[j] == page)
            {
                hit = 1;
                break;
            }
        }

        if(hit == 0)
        {
            frame[pos] = page;
            pos = (pos + 1) % n;
            faults++;
        }

        printf("%d\t", page);

        for(j = 0; j < n; j++)
        {
            if(frame[j] == -1)
                printf("- ");

                printf("%d ", frame[j]);

        if(hit == 0)
            printf("\tPage Fault");
        else
            printf("\tHit");

        printf("\n");
    }

    printf("\nTotal Page Faults = %d\n", faults);

Q.2)  Consider a system with ‘n’ processes and ‘m’ resource types. Accept    number of instances for every resource type. For each process accept the allocation and maximum requirement matrices. Write a program to display the contents of need matrix and to check if the given request of a process can be granted immediately or not. (Use resource request algorithm)    

#include <stdio.h>

int main()
{
    int n, m, i, j, p;
    int allocation[10][10], max[10][10], need[10][10];
    int total[10], available[10], request[10];
    int possible = 1;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    printf("Enter number of resource types: ");
    scanf("%d", &m);

    /* Accept total instances */
    printf("Enter total instances of each resource:\n");
    for(i = 0; i < m; i++)
        scanf("%d", &total[i]);

    /* Accept Allocation */
    printf("\nEnter Allocation Matrix:\n");
    for(i = 0; i < n; i++)
    {
        printf("P%d: ", i);
        for(j = 0; j < m; j++)
            scanf("%d", &allocation[i][j]);
    }

    /* Accept Maximum */
    printf("\nEnter Maximum Matrix:\n");
    for(i = 0; i < n; i++)
    {
        printf("P%d: ", i);
        for(j = 0; j < m; j++)
            scanf("%d", &max[i][j]);
    }

    /* Calculate Available */
    for(j = 0; j < m; j++)
    {
        available[j] = total[j];

        for(i = 0; i < n; i++)
            available[j] -= allocation[i][j];
    }

    /* Calculate Need */
    for(i = 0; i < n; i++)
    {
        for(j = 0; j < m; j++)
            need[i][j] = max[i][j] - allocation[i][j];
    }

    /* Display Need */
    printf("\nNeed Matrix:\n");
    printf("Process\t");

    for(j = 0; j < m; j++)
        printf("R%d ", j);

    for(i = 0; i < n; i++)
    {
        printf("\nP%d\t", i);

        for(j = 0; j < m; j++)
            printf("%d ", need[i][j]);
    }

    /* Display Available */
    printf("\n\nAvailable Resources:\n");
    for(j = 0; j < m; j++)
        printf("%d ", available[j]);

    /* Accept Request */
    printf("\n\nEnter process number for request: ");
    scanf("%d", &p);

    printf("Enter request of P%d:\n", p);
    for(j = 0; j < m; j++)
        scanf("%d", &request[j]);

    /* Check Request <= Need */
    for(j = 0; j < m; j++)
    {
        if(request[j] > need[p][j])
        {
            possible = 0;
            break;
        }
    }

    /* Check Request <= Available */
    if(possible == 1)
    {
        for(j = 0; j < m; j++)
        {
            if(request[j] > available[j])
            {
                possible = 0;
                break;
            }
        }
    }

    if(possible == 1)
        printf("\nRequest can be GRANTED immediately.\n");
    else
        printf("\nRequest CANNOT be granted immediately.\n");

    return 0;
}

Q.1) Write the program to simulate Round Robin (RR) scheduling. The arrival  time and first CPU burst for different n number of processes should be  input to the algorithm. #include <stdio.h>
#include <stdlib.h>
#include <time.h>

int main()
{
    int n, i, currentTime = 0, completed = 0;
    int tq;
    int at[20], bt[20], rem[20], ct[20], wt[20], tat[20];
    int queue[100], front = 0, rear = 0;
    int visited[20] = {0};

    printf("Enter number of processes: ");
    scanf("%d", &n);

    printf("\nEnter Arrival Time and First CPU Burst:\n");

    for(i = 0; i < n; i++)
    {
        printf("P%d Arrival Time: ", i + 1);
        scanf("%d", &at[i]);

        printf("P%d First CPU Burst: ", i + 1);
        scanf("%d", &bt[i]);
    }

    printf("\nEnter Time Quantum: ");
    scanf("%d", &tq);

    srand(time(0));

    /* Generate next CPU burst randomly */
    for(i = 0; i < n; i++)
        bt[i] = bt[i] + rand() % 5;

    for(i = 0; i < n; i++)
        rem[i] = bt[i];

    printf("\nGenerated CPU Burst:\n");

    for(i = 0; i < n; i++)
        printf("P%d = %d\n", i + 1, bt[i]);

    /* Add processes which have already arrived */
    for(i = 0; i < n; i++)
    {
        if(at[i] == 0)
        {
            queue[rear++] = i;
            visited[i] = 1;
        }
    }

    printf("\nGantt Chart:\n");

    while(completed < n)
    {
        /* If queue is empty, move time forward */
        if(front == rear)
        {
            currentTime++;

            for(i = 0; i < n; i++)
            {
                if(visited[i] == 0 && at[i] <= currentTime)
                {
                    queue[rear++] = i;
                    visited[i] = 1;
                }
            }

            continue;
        }

        i = queue[front++];

        printf("| P%d ", i + 1);

        if(rem[i] > tq)
        {
            currentTime = currentTime + tq;
            rem[i] = rem[i] - tq;
        }
        else
        {
            currentTime = currentTime + rem[i];
            rem[i] = 0;

            ct[i] = currentTime;
            tat[i] = ct[i] - at[i];
            wt[i] = tat[i] - bt[i];

            completed++;
        }

        /* Add newly arrived processes */
        int j;
        for(j = 0; j < n; j++)
        {
            if(visited[j] == 0 && at[j] <= currentTime)
            {
                queue[rear++] = j;
                visited[j] = 1;
            }
        }

        /* Put unfinished process back into queue */
        if(rem[i] > 0)
            queue[rear++] = i;
    }

    printf("|\n");

    printf("\nProcess\tAT\tBT\tCT\tTAT\tWT\n");

    float avgWT = 0, avgTAT = 0;
